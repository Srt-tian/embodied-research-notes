    def _make_hook(self, name):
        def hook(module, inputs, output):
            value = output.detach()
            self.records[name] = ActivationStats(
                shape=tuple(value.shape),
                mean=value.mean().item(),
                std=value.std(unbiased=False).item(),
            )
        return hook

    def close(self):
        for handle in self._handles:
            handle.remove()
        self._handles.clear()
